# system= is group:artifact:version of a dependency Maven lists with scope system
# The project declares org.dm:tools:1.0 as system with ${project.basedir}/lib/tools.jar, and org.dm:lib:1.0
# from repo, whose POM asks for org.dm:tools:[2.0]. The hard range wins over the nearer declaration,
# whose scope stays: Maven lists org.dm:tools:jar:2.0:system and reads its jar from repo.
result=success
